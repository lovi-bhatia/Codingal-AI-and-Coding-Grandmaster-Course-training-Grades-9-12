JavaScript Objects, Properties and Math

1. Creating an Object

Code: 
const user = {
  name: "Alex",
  "age": 20,          // quotes optional on valid keys
  active: true
};

console.log(user);
console.log(user.name);
console.log(user["age"]);

// const blocks reassignment, not editing
user.city = "Delhi";
console.log(user.city);

2. Dot vs Bracket Access

Code: 
const car = {
  brand: "Toyota",
  "model year": 2020,
  make: "Corolla"
};

// Dot notation — fixed, known keys
console.log(car.brand);

// Bracket — keys with spaces
console.log(car["model year"]);

// Bracket — dynamic key from a variable
const key = "make";
console.log(car[key]);

// WRONG: car.model year   ← space breaks it
// WRONG: car.key          ← looks for a property named "key"

4. Nested Objects, Shorthand, Methods and this

Code: 
// Nested object (tree)
const order = {
  id: 101,
  customer: { name: "Sam", address: { city: "Mumbai" } }
};
console.log(order.customer.address.city);

// Shorthand (variable name becomes the key)
const name = "Alex";
const age = 20;
const person = { name, age };
console.log(person);

// Computed key
const field = "role";
const account = { [field]: "admin" };
console.log(account.role);

// Method and `this`
const counter = {
  count: 0,
  increment() { this.count++; }
};
counter.increment();
counter.increment();
console.log(counter.count);

5. Looping Over Object Properties

Code: 
const scores = { math: 90, science: 85, english: 78 };

// for...in → keys
for (const key in scores) {
  console.log(key + " = " + scores[key]);
}

// Object.keys / values / entries
console.log(Object.keys(scores));
console.log(Object.values(scores));

// entries + destructuring
for (const [k, v] of Object.entries(scores)) {
  console.log(k, "→", v);
}

// Total via values
const total = Object.values(scores)
  .reduce((sum, n) => sum + n, 0);
console.log("Total: " + total);

6. Object Data Structure — Array of Objects

Code: 
const students = [
  { name: "Alex", marks: 85, city: "Delhi" },
  { name: "Sam",  marks: 72, city: "Mumbai" },
  { name: "Ravi", marks: 91, city: "Delhi" }
];

// Access one record
console.log(students[0].name);

// Filter rows
const topStudents = students.filter(s => s.marks > 80);
console.log(topStudents.map(s => s.name));

// Extract one column
console.log(students.map(s => s.marks));

// Sum one column
const avg = students.reduce((t, s) => t + s.marks, 0)
              / students.length;
console.log("Average: " + avg.toFixed(1));

// Group by a property
const byCity = students.reduce((acc, s) => {
  (acc[s.city] ??= []).push(s.name);
  return acc;
}, {});
console.log(JSON.stringify(byCity));

9. Math.random()

Code: 
// Random decimal: 0 ≤ x < 1
console.log(Math.random());

// Random 0–9 integer
console.log(Math.floor(Math.random() * 10));

// Random integer between min and max (inclusive)
function randomInt(min, max) {
  return Math.floor(Math.random() * (max - min + 1)) + min;
}

console.log(randomInt(1, 6));    // dice
console.log(randomInt(1, 6));
console.log(randomInt(50, 100));

// Pick a random array element
const colors = ["red", "green", "blue"];
const pick = colors[Math.floor(Math.random() * colors.length)];
console.log(pick);

10. The Arguments Object (old way)

Code: 
function total() {
  // `arguments` is auto-created
  console.log("length:", arguments.length);
  console.log("first:", arguments[0]);

  // Sum without knowing the count
  let sum = 0;
  for (let i = 0; i < arguments.length; i++) {
    sum += arguments[i];
  }
  return sum;
}

console.log(total(1, 2, 3));
console.log(total(10, 20, 30, 40));

// Array-like, NOT an array
function check() {
  console.log(Array.isArray(arguments));
  console.log(typeof arguments.map);   // undefined
}
check(1, 2);

// Convert to a real array
function convert() {
  const arr = [...arguments];
  return arr.map(n => n * 2);
}
console.log(convert(1, 2, 3));

11. Rest Parameters — The Modern Replacement

Code: 
// Real array, no conversion needed
function total(...nums) {
  console.log(Array.isArray(nums));
  return nums.reduce((sum, n) => sum + n, 0);
}

console.log(total(1, 2, 3));
console.log(total(10, 20, 30));

// Mix normal + rest (rest must be last)
function tag(label, ...items) {
  return label + ": " + items.join(", ");
}
console.log(tag("Fruits", "apple", "mango", "kiwi"));

// Arrow functions have NO arguments object
const arrow = (...args) => args.length;
console.log(arrow(1, 2, 3, 4));

// Prefer rest (...) in new code.
// Use `arguments` only when reading legacy code.

12. Quick Reference

Concept Syntax Note
Create object const o = { a: 1 }; Key : value pairs
Dot access o.a Fixed key names
Bracket access o["a"] Dynamic / space keys
Add / update o.b = 2; Creates if missing
Delete delete o.a; Removes the property
Check "a" in o Or hasOwnProperty
Nested o.user.address.city Tree structure
Shorthand { name, age } Variables as keys
Computed key { [k]: v } Dynamic property name
Method { greet() { } } Function as property
Keys / values Object.keys(o) Arrays
Loop keys for (const k in o) Objects only
Math constant Math.PI Not a constructor
Round nearest Math.round(4.5) → 5
Round down / up Math.floor() / Math.ceil() Always down / up
Drop decimals Math.trunc(4.9) → 4
Max / min Math.max(...arr) Spread for arrays
Random decimal Math.random() 0 ≤ n < 1
Random integer Math.floor(Math.random()*(max-min+1))+min Inclusive both ends
Arguments object arguments[0] Array-like, not an array
Rest parameters function f(...args) Real array — preferred
