2. Parameters vs Arguments

Code: 
// a, b are PARAMETERS
function add(a, b) {
  return a + b;
}

// 2, 3 are ARGUMENTS
console.log(add(2, 3));

// Too few → missing becomes undefined
console.log(add(5));        // NaN

// Too many → extras ignored
console.log(add(1, 2, 3));  // 3

3. Optional / Default Parameters

Code: 
// Default fires only if arg is undefined
function greet(name = "Guest", role = "user") {
  return `${name} (${role})`;
}

console.log(greet());
console.log(greet("Sam"));
console.log(greet("Sam", "admin"));

// null does NOT trigger the default
console.log(greet(null));

// Default based on earlier param
function area(w, h = w) {
  return w * h;
}
console.log(area(5));     // 5 * 5

4. Rest Parameters (...args)

Code: 
// Collect any number of extra arguments
function sum(...nums) {
  return nums.reduce((total, n) => total + n, 0);
}

console.log(sum(1, 2));         // 3
console.log(sum(1, 2, 3, 4));   // 10
console.log(sum());             // 0

// Can mix with normal params (rest must be last)
function tag(label, ...items) {
  return label + ": " + items.join(", ");
}
console.log(tag("Fruits", "apple", "mango"));

5. Return Values

Code: 
function check(n) {
  if (n < 0) return "negative";  // exits here
  return "positive";             // exits here
  console.log("never runs");     // unreachable
}

console.log(check(-5));
console.log(check(5));

// No return → undefined
function log(msg) {
  console.log(msg);
}
console.log(log("hi"));   // logs "hi" then undefined

6. Scope of Variables in Functions

Code: 
const globalVar = "I am global";

function outer() {
  const outerVar = "I am outer";

  function inner() {
    const innerVar = "I am inner";
    // Can read parent + global
    return [globalVar, outerVar, innerVar].join(" | ");
  }

  // innerVar NOT visible here
  return inner();
}

console.log(outer());

7. Shadowing and Block Scope

Code: 
let color = "red";

function paint() {
  // Shadows the outer 'color'
  let color = "blue";
  return color;
}

console.log(paint());   // blue
console.log(color);     // red (unchanged)

// let / const stay inside { }
{
  let blockOnly = "inside";
}
// console.log(blockOnly);  // Error

// var leaks out of blocks (avoid!)
{
  var leaks = "outside block";
}
console.log(leaks);

8. Nested Functions (Closure Preview)

Code: 
function outerFunction(a) {
  let count = 0;              // lives inside outerFunction

  function innerFunction() {
    count++;                  // inner can modify outer var
    return `${a} called ${count} times`;
  }

  return innerFunction;       // return the inner function
}

const counterA = outerFunction("A");
const counterB = outerFunction("B");

console.log(counterA());
console.log(counterA());
console.log(counterB());  // own copy of count

9. Function Expressions

Code: 
// Anonymous expression
const add = function (a, b) {
  return a + b;
};

// Named expression (name is internal only)
const double = function doubler(n) {
  return n * 2;
};

console.log(add(3, 4));
console.log(double(5));

// NOT hoisted — this would throw:
// console.log(subtract(5, 2));
const subtract = function (a, b) { return a - b; };

console.log(subtract(5, 2));

// Expressions can be passed around (callbacks)
const run = function (fn, value) {
  return fn(value);
};
console.log(run(double, 10));

10. Arrow Functions

Code: 
// Full form
const add = (a, b) => {
  return a + b;
};

// Implicit return (single expression, no braces)
const square = n => n * n;

// No parameters
const sayHi = () => "Hi!";

// Returning an object needs ( )
const makeUser = (name) => ({ user: name });

console.log(add(2, 3));
console.log(square(4));
console.log(sayHi());
console.log(makeUser("Sam"));

// Arrow as callback
[1, 2, 3].forEach(n => console.log(n * 10));

11. Quick Reference

Concept Syntax Note
Declaration function f(a) { return a; } Hoisted
Expression const f = function(a) { return a; }; Not hoisted
Arrow const f = a => a; Short, no own this
Parameter function f(a, b) Placeholder in definition
Argument f(1, 2) Real value passed in call
Default parameter function f(a = 5) Used when arg is undefined
Rest parameter function f(...args) Collects extras into an array
Return return value; Ends function and sends value back
No return function f() {} Returns undefined
Global scope const x = 1; outside fn Visible everywhere
Local scope inside function Dies when function ends
Block scope { let y = 1; } Stays inside { }
Shadowing same name inside fn Hides outer variable locally
Closure nested function + outer var Inner keeps access to outer scope
Callback run(fn, 5) Function passed as a value
