JavaScript Loops

1. The for Loop

Code: 
// Count 1 to 5
for (let i = 1; i <= 5; i++) {
  console.log("i =", i);
}

// Count backwards
for (let i = 5; i >= 1; i--) {
  console.log("down:", i);
}

// Step by 2
for (let i = 0; i <= 10; i += 2) {
  console.log("step:", i);
}

2. The while Loop

Code: 
let i = 1;

while (i <= 5) {
  console.log("i =", i);
  i++;                     // must update yourself
}

// Halving until value is small
let n = 100;
while (n > 10) {
  console.log("n =", n);
  n = n / 2;
}

// Guarding against infinite loops:
// every while body must move the
// condition toward false.

3. The do...while Loop

Code: 
// Runs once even when condition is false
let a = 10;

do {
  console.log("runs once, a =", a);
  a++;
} while (a < 5);

// Normal use
let count = 0;
do {
  console.log("count =", count);
  count++;
} while (count < 3);

4. for...of — Values of an Array or String

Code: 
const fruits = ["apple", "mango", "kiwi"];

for (const fruit of fruits) {
  console.log(fruit);
}

// Loop a string
for (const ch of "abc") {
  console.log("char:", ch);
}

// With index using .entries()
for (const [i, fruit] of fruits.entries()) {
  console.log(i, fruit);
}

5. for...in — Keys of an Object

Code: 
const user = {
  name: "Alex",
  age: 20,
  city: "Delhi"
};

for (const key in user) {
  console.log(key + ": " + user[key]);
}

// Count properties
let total = 0;
for (const k in user) total++;
console.log("properties:", total);

6. break and continue

Code: 
// break — stop the loop
for (let i = 1; i <= 10; i++) {
  if (i === 5) break;
  console.log("break test:", i);
}

// continue — skip this iteration
for (let i = 1; i <= 6; i++) {
  if (i % 2 === 0) continue;
  console.log("odd:", i);
}

// Combined search pattern
const nums = [3, 8, 12, 15, 20];
let firstBig;
for (const n of nums) {
  if (n > 10) { firstBig = n; break; }
}
console.log("first > 10:", firstBig);

7. Nested Loops

Code: 
// Outer x inner combinations
for (let i = 1; i <= 3; i++) {
  for (let j = 1; j <= 3; j++) {
    console.log(i + " x " + j + " = " + (i * j));
  }
}

// Triangle pattern
let star = "";
for (let i = 1; i <= 4; i++) {
  star += "*";
  console.log(star);
}

8. Common Loop Patterns

Code: 
const data = [4, 8, 15, 16, 23, 42];

// SUM
let sum = 0;
for (const n of data) sum += n;

// COUNT
let evens = 0;
for (const n of data) if (n % 2 === 0) evens++;

// MAX
let max = data[0];
for (const n of data) if (n > max) max = n;

// BUILD a string
let text = "";
for (const n of data) text += n + " ";

console.log("sum   :", sum);
console.log("evens :", evens);
console.log("max   :", max);
console.log("text  :", text.trim());

9. forEach — The Array Method Version

Code: 
const items = ["pen", "book", "bag"];

items.forEach(function (item, index) {
  console.log(index, item);
});

// Arrow version
items.forEach((item, i) => console.log(i + ": " + item));

// Same result as for...of, but:
// - cannot use break or continue
// - is a method, not a loop keyword

10. Quick Reference

Loop Syntax Best For
for for (let i = 0; i < n; i++) { } Known number of iterations
while while (cond) { } Unknown count, check first
do...while do { } while (cond); Must run at least once
for...of for (const x of arr) { } Array / string values
for...in for (const k in obj) { } Object keys
break if (cond) break; Exit the loop early
continue if (cond) continue; Skip rest of this iteration
Nested loop inside loop Grids, tables, combinations
forEach arr.forEach(fn) Clean array loop (no break)
Infinite loop condition never flips Bug — always update your counter
