1. Creating Arrays

Code: 
const nums = [10, 20, 30];

2. Index, Length, at()

Code: 
const fruits = ["apple", "mango", "kiwi", "pear"];

console.log(fruits[0]);                  // first
console.log(fruits[2]);                  // third
console.log(fruits[fruits.length - 1]);  // last

console.log(fruits.at(-2));              // second last
console.log(fruits.length);

3. Adding and Removing — End and Start

Code: 
const arr = [1, 2, 3];

arr.push(4);        // add to end
console.log(arr)    // [1,2,3,4]

arr.pop();          // remove from end
console.log(arr)    // [1,2,3]

arr.unshift(0);     // add to start
console.log(arr)    // [0,1,2,3]

arr.shift();        // remove from start
console.log(arr)    // [1,2,3]

// Return values
console.log(arr.push(9));   // new length
console.log(arr.pop());     // removed item

// These all MUTATE the original array.
Output:

4. splice vs slice

Code: 
// SPLICE — mutates
const a = ["a", "b", "c", "d", "e"];
a.splice(2, 1);          // remove 1 at index 2
console.log(a);          // [a,b,d,e]

a.splice(1, 0, "x");     // insert "x" at 1
console.log(a);          // [a,x,b,d,e]

a.splice(2, 2, "y");     // replace 2 items
console.log(a);          // [a,x,y,e]

// SLICE — does not mutate
const b = ["a", "b", "c", "d", "e"];
const part = b.slice(1, 3);
console.log(part);       // [b,c]
console.log(b);          // full array unchanged

console.log(b.slice(-2)); // last two
Output:

5. sort and reverse

Code: 
// Strings sort naturally
const names = ["Sam", "Alex", "Ravi"];
names.sort();
console.log(names);
names.reverse();
console.log(names);

// Numbers need a compare function
const nums = [40, 5, 100, 25];
nums.sort();                       // wrong: sorts as text
console.log(nums);

// Correct numeric sort
const nums2 = [40, 5, 100, 25];
nums2.sort((a, b) => a - b);       // ascending
console.log(nums2);
nums2.sort((a, b) => b - a);       // descending
console.log(nums2);

// Keep original safe with spread
const safe = [...nums2].sort((a, b) => a - b);

Code:

  { id: 1, name: "Alex" },

8. map and filter

Code: 
const nums = [1, 2, 3, 4, 5];

// MAP — same length, transformed
const doubled = nums.map(n => n * 2);
console.log(doubled);
console.log(nums);              // original safe

// Transform to strings
const labels = nums.map(n => "item-" + n);
console.log(labels);

// FILTER — shorter array
const evens = nums.filter(n => n % 2 === 0);
console.log(evens);

// Chain map and filter
const result = nums
  .filter(n => n > 2)
  .map(n => n * 10);
console.log(result);

// Array of objects
const users = [
  { name: "Alex", age: 20 },
  { name: "Sam",  age: 30 },
  { name: "Ravi", age: 25 }
];
console.log(users.map(u => u.name));
console.log(users.filter(u => u.age >= 25));
Output:

9. reduce — Collapse to a Single Value

Code: 
const nums = [1, 2, 3, 4, 5];

// Sum
const sum = nums.reduce((acc, n) => acc + n, 0);
console.log("sum:", sum);

// Product
const product = nums.reduce((acc, n) => acc * n, 1);
console.log("product:", product);

// Max
const max = nums.reduce((a, b) => a > b ? a : b);
console.log("max:", max);

// Count occurrences
const words = ["a", "b", "a", "c", "a"];
const counts = words.reduce((acc, w) => {
  acc[w] = (acc[w] || 0) + 1;
  return acc;
}, {});
console.log(counts);

// ALWAYS pass an initial value (second arg).
// Without it, reduce uses the first item as acc.

10. some and every

Code: 
const ages = [15, 22, 30, 18];

// some — is at least one true?
console.log(ages.some(a => a >= 21));
console.log(ages.some(a => a > 100));

// every — are all true?
console.log(ages.every(a => a >= 18));
console.log(ages.every(a => a > 10));

// Real use: validate a form
const inputs = ["Sam", "sam@site.com", ""];
const hasEmpty = inputs.some(v => v.trim() === "");
console.log("form has empty field:", hasEmpty);

// some / every stop early once decided.

11. join, concat, flat, flatMap

Code: 
// join — array → string
const parts = ["2025", "10", "28"];
console.log(parts.join("-"));
console.log(parts.join(""));
console.log(["a", "b"].join(" / "));

// concat — merge arrays
const a = [1, 2];
const b = [3, 4];
console.log(a.concat(b));
console.log([...a, ...b]);

// flat — remove nesting
const nested = [1, [2, [3, [4]]]];
console.log(nested.flat());
console.log(nested.flat(2));
console.log(nested.flat(Infinity));

// flatMap — map then flat one level
const phrases = ["hi there", "good day"];
console.log(phrases.flatMap(p => p.split(" ")));

12. Spread (...) and Destructuring

Code: 
const original = [1, 2, 3];

// Copy
const copy = [...original];
copy.push(99);
console.log(copy);
console.log(original);   // unchanged

// Merge
const merged = [...original, 4, 5];
console.log(merged);

// Spread into Math functions
const nums = [12, 45, 7];
console.log(Math.max(...nums));

// Destructuring
const [first, second, ...rest] = [10, 20, 30, 40];
console.log(first, second, rest);

// Skip an item with a comma
const [a, , c] = ["x", "y", "z"];
console.log(a, c);

// Default values
const [p = "empty"] = [];
console.log(p);

// Swap without a temp
let x = 1, y = 2;
[x, y] = [y, x];
console.log(x, y);

13. Common Patterns

Code: 
const data = [4, 8, 15, 16, 23, 42];

// SUM
console.log(data.reduce((s, n) => s + n, 0));

// AVERAGE
const avg = data.reduce((s, n) => s + n, 0) / data.length;
console.log(avg.toFixed(2));

// MAX
console.log(Math.max(...data));

// UNIQUE
const dupes = [1, 2, 2, 3, 3, 3];
console.log([...new Set(dupes)]);

// NUMERIC SORT
console.log([...data].sort((a, b) => a - b));

// REVERSE without touching original
console.log([...data].reverse());

// CHECK EMPTY
console.log(data.length === 0);

// ARRAY OF OBJECTS — get a column
const users = [
  { name: "Alex", score: 90 },
  { name: "Sam",  score: 75 },
  { name: "Ravi", score: 88 }
];
console.log(users.map(u => u.name));
console.log(users.filter(u => u.score > 80).map(u => u.name));
console.log(users.sort((a, b) => b.score - a.score)[0].name);

14. Checking for Arrays

Code: 
console.log(Array.isArray([]));           // true
console.log(Array.isArray("hi"));          // false
console.log(Array.isArray({ length: 2 })); // false
console.log(Array.isArray(new Array()));   // true

// typeof is not helpful for arrays
console.log(typeof []);         // "object"

// Only use Array.isArray — it is reliable
// across different windows / iframes,
// whereas instanceof can fail.

15. Quick Reference

Method Does Mutates? Returns
push(x) Add to end Yes New length
pop() Remove from end Yes Removed item
unshift(x) Add to start Yes New length
shift() Remove from start Yes Removed item
splice(i, n) Remove / insert Yes Removed items
sort(fn) Sort in place Yes Same array
reverse() Reverse in place Yes Same array
fill(v) Fill with value Yes Same array
slice(a, b) Copy a range No New array
concat(arr) Merge arrays No New array
join(sep) Array → string No String
indexOf(x) First index of x No Index or -1
includes(x) Contains x No true / false
forEach(fn) Loop each item No undefined
map(fn) Transform items No New array
filter(fn) Keep matching No New array
find(fn) First match No Item or undefined
findIndex(fn) Index of first match No Index or -1
reduce(fn, init) Collapse to one value No Accumulated value
some(fn) Any pass No true / false
every(fn) All pass No true / false
flat(depth) Flatten nesting No New array
flatMap(fn) Map then flat 1 No New array
Array.isArray(x) Is it an array? No true / false
