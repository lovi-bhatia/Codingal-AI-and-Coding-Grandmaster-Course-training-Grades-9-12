HTML DOM and DOM Structure

1. What Is the DOM?

The DOM is the browser's in-memory tree of nodes built from your HTML. JavaScript reads and edits this tree, and the screen updates instantly.

Code: 
// document is the root of the DOM
console.log(document.title);
console.log(document.doctype ? "has doctype" : "no doctype");
console.log(document.documentElement.tagName);
console.log(document.body.tagName);

// The DOM has MORE nodes than you wrote
// (whitespace text nodes, comments, etc.)
console.log(document.body.childNodes.length);

2. DOM Structure — The Tree

HTML source: 
<html>
  <head>
    <title>My Page</title>
  </head>
  <body>
    <h1>Hello</h1>
    <ul>
      <li>One</li>
      <li>Two</li>
    </ul>
  </body>
</html>
DOM tree it becomes:
document └─ html ├─ head │ └─ title │ └─ "My Page" └─ body ├─ h1 │ └─ "Hello" └─ ul ├─ li ─ "One" └─ li ─ "Two"

4. Node Types — Everything Is a Node

Code: 
const target = document.getElementById("nodeTarget");

console.log("Element node :", target.nodeName,
            "| type:", target.nodeType);

console.log("Text node    :",
            target.firstChild.nodeType);   // 3

console.log("children (elements):",
            target.children.length);

console.log("childNodes (everything):",
            target.childNodes.length);

// 1 = Element, 3 = Text, 8 = Comment, 9 = Document
for (const n of target.childNodes) {
  console.log(n.nodeType, JSON.stringify(n.nodeName));

5. Selecting Elements

Code: 
// by id (fastest)
const byId = document.getElementById("sel-personal");

// by class (LIVE collection)
const byClass = document.getElementsByClassName("sel-item");

// by tag (LIVE collection)
const byTag = document.getElementsByTagName("li");

// CSS selector — first match or null
const first = document.querySelector(".sel-item");

// CSS selector — all matches, STATIC NodeList
const all = document.querySelectorAll(".sel-item");

// Compound selectors
const special = document.querySelectorAll(
  ".sel-item[data-active='true']"
);

console.log(byId.textContent.trim());
console.log("class count:", byClass.length);
console.log("tag count:", byTag.length);
console.log("first match:", first.textContent);
console.log("all count:", all.length);
console.log("special count:", special.length);

9. Creating, Inserting and Removing Nodes

Code: 
const list = document.getElementById("makeList");

// 1. Create
const li = document.createElement("li");
li.textContent = "New item";
li.classList.add("list-group-item");

// 2. Insert
list.appendChild(li);              // add at end

const first = document.createElement("li");
first.textContent = "Prepended item";
first.className = "list-group-item";
list.prepend(first);               // add at start

// 3. Insert before / after a sibling
const ref = list.children[1];
const extra = document.createElement("li");
extra.className = "list-group-item";
extra.textContent = "Inserted before item 2";
ref.before(extra);

// 4. Remove
list.lastElementChild.remove();

// 5. Bulk insert with a DocumentFragment (one reflow)
const frag = document.createDocumentFragment();
["Batch 1", "Batch 2", "Batch 3"].forEach(t => {
  const item = document.createElement("li");
  item.className = "list-group-item";
  item.textContent = t;
  frag.appendChild(item);
});
list.appendChild(frag);

// 6. Empty a container
// list.innerHTML = "";

10. Classes and Styles

Code: 
const box = document.getElementById("styleBox");

// Classes — the preferred way
box.classList.add("bg-warning");
box.classList.remove("bg-light");
box.classList.toggle("text-dark");
console.log("has bg-warning :", box.classList.contains("bg-warning"));

// Inline styles (camelCase for CSS kebab-case)
box.style.padding = "20px";
box.style.borderRadius = "8px";
box.style.letterSpacing = "0.5px";   // letter-spacing
box.style.cssText = "padding: 18px; border-radius: 8px;";

// Read computed style (final, after CSS is applied)
const styles = getComputedStyle(box);
console.log("background:", styles.backgroundColor);
console.log("padding   :", styles.padding);

11. Events — Reacting to the DOM

Code: 
const btn = document.getElementById("eventBtn");
const out = document.getElementById("eventOutput");

// attach a listener (do NOT use onclick="" in HTML)
btn.addEventListener("click", function (event) {
  event.target.textContent = "Clicked!";
  event.target.classList.add("btn-success");
  console.log("button clicked");
});

// SINGLE ELEMENT events
const input = document.getElementById("eventInput");
input.addEventListener("input", (e) => {
  document.getElementById("liveInput").textContent =
    "You typed: " + e.target.value;
});

// EVENT DELEGATION — one listener for many children
document.getElementById("delegateList")
  .addEventListener("click", (e) => {
    if (e.target.tagName === "BUTTON") {
      e.target.previousElementSibling.textContent =
        "clicked: " + e.target.textContent;
    }
  });

Siblings el.nextElementSibling Skips text nodes
