JavaScript Syntax, Guide and Commenting

1. Variables and Data Types

Code: 
const name = "Alex";        // String  (fixed)
let age = 20;               // Number  (changeable)
const active = true;        // Boolean
const marks = [85, 90, 78]; // Array
const student = {           // Object
  city: "Delhi",
  year: 2
};

age = 21;                   // let can be reassigned

console.log(name, age, active);
console.log(marks[0]);
console.log(student.city);

3. Operators

Code: 
let a = 10, b = 3;

console.log(a + b);      // 13
console.log(a - b);      // 7
console.log(a * b);      // 30
console.log(a % b);      // 1

console.log(a > b);      // true
console.log(a === "10"); // false (type differs)
console.log(a == "10");  // true  (avoid ==)

4. Commenting

Code: 
// Single-line comment — explains one line

/*
  Multi-line comment — explains a bigger block
  or temporarily disables code
*/

/**
 * Adds two numbers.
 * @param {number} a - first number
 * @param {number} b - second number
 * @returns {number} sum
 */
function add(a, b) {
  return a + b;   // returns the result
}

console.log(add(4, 6));

5. Functions

Code: 
// Function declaration
function greet(name) {
  return `Hi ${name}`;
}

// Arrow function (shorter)
const square = n => n * n;

console.log(greet("Ravi"));
console.log(square(5));
