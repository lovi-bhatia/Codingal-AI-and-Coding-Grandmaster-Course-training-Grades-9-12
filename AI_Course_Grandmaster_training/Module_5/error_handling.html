Error Handling in JavaScript

Rule to memorise: try the risky operation, catch only what you can handle, finally for cleanup. Always throw new Error(...) — never a string.

1. Built-in Error Types

Code: 
// ReferenceError — variable does not exist
try {
  console.log(nonExistent);
} catch (e) {
  console.log(e.name, "|", e.message);
}

// TypeError — wrong type / reading on null
try {
  null.name;
} catch (e) {
  console.log(e.name, "|", e.message);
}

// RangeError — out of range
try {
  new Array(-1);
} catch (e) {
  console.log(e.name, "|", e.message);
}

// SyntaxError — from eval or JSON.parse
try {
  JSON.parse("{bad}");
} catch (e) {
  console.log(e.name, "|", e.message);
}

// URIError
try {
  decodeURIComponent("%");
} catch (e) {
  console.log(e.name, "|", e.message);
}

// All inherit from Error
console.log(e instanceof Error);   // true

2. Basic try / catch

Code: 
// No error → catch skipped
try {
  const x = 10 / 2;
  console.log("result:", x);
} catch (err) {
  console.log("never runs");
}

// Error → catch runs, control jumps here
try {
  const user = null;
  console.log(user.name);
  console.log("this line is skipped");
} catch (err) {
  console.log("Caught:", err.name);
  console.log("Message:", err.message);
}

// Execution continues after the try/catch
console.log("program continues");

3. finally — Always Runs

Code: 
// Runs after success
try {
  console.log("try succeeds");
} finally {
  console.log("finally runs (success)");
}

// Runs after a caught error
try {
  throw new Error("boom");
} catch (e) {
  console.log("caught:", e.message);
} finally {
  console.log("finally runs (error)");
}

// Runs even when the function returns
function test() {
  try {
    return "from try";
  } finally {
    console.log("finally runs before the return");
  }
}
console.log(test());

// Cleanup pattern
function withSpinner() {
  let spinner = "visible";
  try {
    return "data loaded";
  } finally {
    spinner = "hidden";
    console.log("spinner:", spinner);
  }
}
console.log(withSpinner());

4. The Error Object

Code: 
function throwIt() {
  throw new TypeError("expected a number");
}

try {
  throwIt();
} catch (err) {
  console.log("name    :", err.name);
  console.log("message :", err.message);
  console.log("has stack:", typeof err.stack === "string");
  console.log("first stack line:");
  console.log("  " + err.stack.split("\n")[0].trim());

  console.log("is Error     :", err instanceof Error);
  console.log("is TypeError :", err instanceof TypeError);
}

// Attaching a cause (ES2022)
try {
  try {
    JSON.parse("{bad}");
  } catch (inner) {
    throw new Error("config could not be parsed", { cause: inner });
  }
} catch (outer) {
  console.log("");
  console.log("outer message:", outer.message);
  console.log("cause name   :", outer.cause.name);

5. throw — Always Throw an Error

Code: 
// ---- GOOD ----
try {
  throw new Error("something went wrong");
} catch (e) {
  console.log("Error object:", e.message);
  console.log("has stack   :", typeof e.stack === "string");
}

// ---- LEGAL BUT BAD ----
try {
  throw "a string";              // no stack, no name
} catch (e) {
  console.log("");
  console.log("string thrown:", e);
  console.log("typeof       :", typeof e);
  console.log("e.stack      :", typeof e.stack);
}

// ---- Rethrowing inside catch ----
function risky() {
  throw new Error("inner error");
}

try {
  try {
    risky();
  } catch (e) {
    console.log("");
    console.log("first catch  :", e.message);
    throw e;                     // rethrow the SAME error
  }
} catch (e) {
  console.log("outer catch  :", e.message);
}

// ---- Guard-first style (preferred) ----
function divide(a, b) {
  if (typeof a !== "number") throw new TypeError("a must be a number");
  if (b === 0)               throw new RangeError("b cannot be zero");
  return a / b;
}
console.log("");
console.log("divide(10, 2)  →", divide(10, 2));

6. Custom Error Classes

Code: 
class ValidationError extends Error {
  constructor(field, message) {
    super(message);               // MUST call super first
    this.name  = "ValidationError";
    this.field = field;
  }
}

class NotFoundError extends Error {
  constructor(resource) {
    super(resource + " not found");
    this.name = "NotFoundError";
    this.resource = resource;
  }
}

// Usage
function createUser(data) {
  if (!data.name) {
    throw new ValidationError("name", "name is required");
  }
  if (data.name.length < 2) {
    throw new ValidationError("name", "name is too short");
  }
  return { id: 1, name: data.name };
}

try {
  createUser({ name: "" });
} catch (e) {
  console.log("name    :", e.name);
  console.log("message :", e.message);
  console.log("field   :", e.field);
  console.log("instanceof ValidationError:", e instanceof ValidationError);
  console.log("instanceof Error         :", e instanceof Error);
}

// ---- Wrapping errors with context ----
function loadConfig(text) {
  try {
    return JSON.parse(text);
  } catch (e) {
    throw new Error("invalid config", { cause: e });
  }
}

try {
  loadConfig("{bad}");
} catch (e) {
  console.log("");
  console.log("wrapped  :", e.message);
  console.log("cause    :", e.cause.name);
}

7. Catching Specific Types with instanceof

Code: 
class ValidationError extends Error {
  constructor(msg) { super(msg); this.name = "ValidationError"; }
}

function handle(err) {
  // Check the MOST SPECIFIC class first
  if (err instanceof ValidationError) {
    return "validation: " + err.message;
  }
  if (err instanceof TypeError) {
    return "type error: " + err.message;
  }
  if (err instanceof RangeError) {
    return "range error: " + err.message;
  }
  if (err instanceof Error) {
    return "generic: " + err.message;
  }
  return "non-error thrown: " + String(err);
}

// Same catch, different subtypes
try { throw new ValidationError("missing email"); }
catch (e) { console.log(handle(e)); }

try { throw new TypeError("not a function"); }
catch (e) { console.log(handle(e)); }

try { throw new RangeError("too big"); }
catch (e) { console.log(handle(e)); }

// Rethrow anything you cannot handle
try {
  throw new Error("unknown origin");
} catch (e) {
  if (e instanceof ValidationError) {
    console.log("handle here");
  } else {
    console.log("rethrowing:", e.message);
  }
}

8. Async Errors — await and Promises

Code: 
// Promises that reject
const badFetch = () =>
  Promise.reject(new Error("network down"));

// ---- await + try/catch (caught) ----
async function loadA() {
  try {
    const data = await badFetch();
    return data;
  } catch (err) {
    return "handled: " + err.message;
  } finally {
    // runs here too
  }
}

// ---- .catch on the chain ----
function loadB() {
  return badFetch()
    .then(d => d)
    .catch(err => "handled: " + err.message)
    .finally(() => console.log("B finally runs"));
}

// ---- async fn turns throw into a rejected promise ----
async function loadC() {
  throw new Error("async boom");   // becomes a rejection
}
loadC().catch(e => console.log("C caught:", e.message));

// ---- unhandled rejection demo ----
window.addEventListener(
  "unhandledrejection",
  e => console.log("unhandled:", e.reason.message),
  { once: true }
);
Promise.reject(new Error("forgotten"));
console.log("sync code runs first");

// Run everything
loadA().then(v => console.log("A:", v));
loadB().then(v => console.log("B:", v));

9. Global Safety Nets

Last-resort handlers for anything that slipped through. Useful for monitoring, not for normal flow.

Code: 
// UNCAUGHT ERRORS
window.addEventListener("error", (event) => {
  console.log("global error event");
  console.log("  message :", event.message);
  console.log("  file    :", event.filename);
  console.log("  line    :", event.lineno);
  console.log("  error   :", event.error?.name);
});

// Trigger an uncaught error
setTimeout(() => {
  throw new Error("uncaught demo");
}, 0);

// UNHANDLED PROMISE REJECTIONS
window.addEventListener("unhandledrejection", (event) => {
  console.log("unhandled rejection");
  console.log("  reason  :", event.reason?.message);
  event.preventDefault();          // stop the console warning
});

setTimeout(() => {
  Promise.reject(new Error("forgotten promise"));
}, 100);

// USE FOR:
//   - logging to a server
//   - showing a fallback UI
//   - catching bugs during development
// NOT FOR:
//   - expected errors that should be handled locally

10. Practical Patterns

Code: 
// A. GUARD CLAUSES — throw early
function divide(a, b) {
  if (typeof a !== "number") throw new TypeError("a must be a number");
  if (typeof b !== "number") throw new TypeError("b must be a number");
  if (b === 0)               throw new RangeError("b cannot be zero");
  return a / b;
}

// B. SAFE PARSE — best-effort conversion
function safeParse(text, fallback = {}) {
  try {
    return JSON.parse(text);
  } catch {
    return fallback;
  }
}
console.log("parse valid:  ", safeParse('{"a":1}'));
console.log("parse invalid:", safeParse("bad json"));

// C. WRAP WITH CONTEXT
function readSetting(key) {
  try {
    return localStorage.getItem(key);
  } catch (e) {
    throw new Error("could not read " + key, { cause: e });
  }
}

// D. RETRY — handle transient failures
async function retry(fn, attempts = 3) {
  let lastErr;
  for (let i = 1; i <= attempts; i++) {
    try {
      return await fn();
    } catch (e) {
      lastErr = e;
      console.log("  attempt " + i + " failed:", e.message);
    }
  }
  throw lastErr;
}

// E. LOG ONCE — at the point of handling
function safeDivide(a, b) {
  try {
    return divide(a, b);
  } catch (e) {
    console.warn("[safeDivide]", e.name + ": " + e.message);
    return null;
  }
}

console.log("divide(10, 2)  →", divide(10, 2));
console.log("safeDivide(10, 0) →", safeDivide(10, 0));

// Run retry example
let tries = 0;
retry(async () => {
  tries++;
  if (tries < 3) throw new Error("flaky");
  return "succeeded on attempt " + tries;
}).then(v => console.log("retry result:", v));

11. Anti-Patterns — What Not To Do

Code: 
// BAD 1 — swallowing silently
function swallow() {
  try { null.x; } catch (e) { /* nothing */ }
}
swallow();
console.log("BUG 1: error was silently lost");

// BAD 2 — throwing a string (no stack)
try {
  throw "just a string";
} catch (e) {
  console.log("BUG 2: typeof err = " + typeof e + ", no .stack");
}

// BAD 3 — catch-and-rethrow with no new information
function pointless() {
  try { null.x; } catch (e) { throw e; }   // same as no try
}
try { pointless(); } catch (e) {
  console.log("BUG 3: try/catch added nothing");
}

// BAD 4 — return inside finally swallows the result
function tricky() {
  try {
    return "try result";
  } finally {
    return "finally always wins";   // overrides try's return
  }
}
console.log("BUG 4:", tricky());

// BAD 5 — try/catch for expected control flow
function badCheck(value) {
  try {
    return JSON.parse(value);
  } catch {
    return null;
  }
}
// Better: guard first when there is a cheap check
function betterCheck(value) {
  if (typeof value !== "string") return null;
  if (!value.trim().startsWith("{")) return null;
  try { return JSON.parse(value); } catch { return null; }
}
console.log("parse attempt 1:", badCheck('{"a":1}'));
console.log("parse attempt 2:", badCheck("not json"));

12. The Result Pattern — Alternative to Throwing

When an error is an expected outcome (validation, parsing), return a box instead of throwing.

Code: 
// Return { ok: true, value } or { ok: false, error }
function parseNumber(str) {
  if (typeof str !== "string") {
    return { ok: false, error: "input is not a string" };
  }
  const n = Number(str);
  if (!Number.isFinite(n)) {
    return { ok: false, error: "not a valid number" };
  }
  return { ok: true, value: n };
}

// CALLER decides how to react — no exceptions in the flow
const a = parseNumber("42");
console.log(a);
if (a.ok) console.log("value:", a.value);
else      console.log("error:", a.error);

const b = parseNumber("42px");
console.log(b);
if (!b.ok) console.log("error:", b.error);

// Throwing version — for contrast
function parseNumberOrThrow(str) {
  const n = Number(str);
  if (!Number.isFinite(n)) throw new TypeError("not a number");
  return n;
}

try {
  console.log("throw version:", parseNumberOrThrow("42"));
} catch (e) {
  console.log("never runs");
}

// WHEN TO USE WHICH
// result pattern → validation, parsing, expected failures
// throw          → unexpected states, missing dependencies,
//                  anything the caller probably cannot continue past

13. Quick Reference

Error Types

Type When It Throws Example
SyntaxError Invalid syntax JSON.parse("{bad}")
ReferenceError Variable does not exist console.log(unknown)
TypeError Wrong type or not callable null.x, notFn()
RangeError Number out of allowed range new Array(-1)
URIError Malformed URI decodeURIComponent("%")
EvalError Historical — rarely thrown —
AggregateError Multiple errors in one Promise.any failures

Skeleton

Part Runs When Use For
try Always The risky operation
catch Only on throw / reject Handle, log, or rethrow
finally Always (even on return) Cleanup — restore UI, close

Error Object

Property Meaning
err.name "TypeError", "ValidationError", …
err.message Human-readable description
err.stack Origin trace
err.cause Underlying error (ES2022)
err instanceof Error true for standard and custom
err instanceof TypeError true for TypeError only

Rules

Rule Why
Always throw new Error(...) Gets name, message, and stack
Never throw strings or plain objects No stack, no instanceof
Check instanceof in order specific → general Subclasses match parents
Rethrow what you cannot handle Prevents silent failures
Never return inside finally Overrides try/catch and swallows errors
Wrap async in try/catch around await Rejections become catches
Attach .catch to every promise chain Otherwise unhandledrejection fires
Log once — where it is handled Avoid duplicate noise
Use cause when wrapping Keeps the original error
Handle expected failures with the result pattern Exceptions are for genuinely unusual states
