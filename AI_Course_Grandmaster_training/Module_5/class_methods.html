1. Class Basics

A class groups data and behaviour into a reusable blueprint. Create objects from it with new.

Code: 
class User {
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }
  greet() {
    return "Hi " + this.name;
  }
  describe() {
    return this.name + " (" + this.age + ")";
  }
}

const alex = new User("Alex", 20);
const sam  = new User("Sam", 25);

console.log(alex.name);
console.log(alex.greet());
console.log(sam.describe());

// Two instances = two independent objects
console.log(alex instanceof User);
console.log(typeof User);         // "function"

2. Class Anatomy — All the Parts

Code: 
class Product {
  // 1. static field (one per class)
  static CATEGORY = "electronics";

  // 2. instance field (one per object)
  qty = 1;

  constructor(name, price) {
    this.name = name;
    this.price = price;
  }

  // 3. instance method
  total() {
    return this.price * this.qty;
  }

  // 4. static method
  static describe() {
    return "Shop category: " + Product.CATEGORY;
  }

  // 5. getter / setter
  get label() { return this.name.toUpperCase(); }
  set label(v) { this.name = v.toLowerCase(); }
}

const p = new Product("Lamp", 500);

console.log(p.name);
console.log(p.total());
console.log(Product.describe());
console.log(p.label);
p.label = "DESK";
console.log(p.name);

3. Instance vs Static Members

Instance members belong to each object. Static members belong to the class itself.

Code: 
class Counter {
  static total = 0;      // shared by ALL instances

  constructor(label) {
    this.label = label;         // per-object
    this.count = 0;             // per-object
    Counter.total++;            // increment the shared counter
  }

  hit() {
    this.count++;
    return `${this.label}: ${this.count}`;
  }

  // static: no `this` of an instance
  static summary() {
    return "instances created: " + Counter.total;
  }
}

const a = new Counter("A");
const b = new Counter("B");

console.log(a.hit());
console.log(a.hit());
console.log(b.hit());
console.log(Counter.summary());

// Where these live:
console.log(typeof a.hit);        // function (via prototype)
console.log(typeof a.summary);    // undefined (static is on the class only)
console.log(typeof Counter.hit);  // undefined (methods need an instance)

4. Static Factory Method — the Classic Use Case

A static method that builds instances for you. Used everywhere in real code (e.g. Date.now(), Array.from()).

Code: 
class User {
  constructor(name, role) {
    this.name = name;
    this.role = role;
  }

  static fromJSON(json) {
    const data = typeof json === "string" ? JSON.parse(json) : json;
    return new User(data.name, data.role);
  }

  static guest() {
    return new User("Guest", "viewer");
  }

  describe() {
    return this.name + " (" + this.role + ")";
  }
}

const a = new User("Alex", "admin");
const b = User.fromJSON('{"name":"Sam","role":"editor"}');
const c = User.guest();

console.log(a.describe());
console.log(b.describe());
console.log(c.describe());

// Why not just `new User(...)` everywhere?
// `fromJSON` handles parsing; callers stay clean.

5. Inheritance with extends

Code: 
class Animal {
  constructor(name) {
    this.name = name;
  }
  speak() { return this.name + " makes a sound"; }
  intro() { return "I am " + this.name; }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);              // must call BEFORE this
    this.breed = breed;
  }
  speak() {
    // override + extend using super
    return this.name + " barks (" + this.breed + ")";
  }
}

class Puppy extends Dog {
  constructor(name, breed) {
    super(name, breed);
    this.age = 0;
  }
  intro() {
    // call parent version, then add more
    return super.intro() + ", a puppy";
  }
}

const d = new Dog("Rex", "Labrador");
const p = new Puppy("Buddy", "Beagle");

console.log(d.intro());
console.log(d.speak());
console.log(p.intro());     // overridden
console.log(p.speak());     // inherited 3 levels up

// instanceof walks the whole chain
console.log(p instanceof Puppy);
console.log(p instanceof Dog);
console.log(p instanceof Animal);
console.log(d instanceof Puppy);

6. Static Members Are Inherited Too

A subclass automatically inherits its parent's static methods and fields — but not from the prototype chain, from the class chain.

Code: 
class Shape {
  static TYPE = "shape";
  static count = 0;

  static register() {
    Shape.count++;
    return "registered " + Shape.count;
  }

  area() { return 0; }
}

class Circle extends Shape {
  constructor(r) { super(); this.r = r; }
  area() { return Math.PI * this.r * this.r; }

  // subclass static override
  static register() {
    return "circle: " + super.register();
  }
}

const c = new Circle(2);

// Instance method
console.log(c.area().toFixed(2));

// Static inherited
console.log(Circle.TYPE);
console.log(Circle.register());
console.log(Shape.register());

// Static lives on the class
console.log(typeof c.register);     // undefined

7. Private Fields and Methods (#)

# members are truly private — no outside access, even with bracket notation.

Code: 
class Bank {
  #balance = 0;              // private field
  #history = [];             // private field

  #log(kind, amt) {          // private method
    this.#history.push({ kind, amt });
  }

  deposit(amt) {
    if (amt <= 0) throw new Error("invalid amount");
    this.#balance += amt;
    this.#log("deposit", amt);
    return this.#balance;
  }

  withdraw(amt) {
    if (amt > this.#balance) throw new Error("insufficient");
    this.#balance -= amt;
    this.#log("withdraw", amt);
    return this.#balance;
  }

  get balance() { return this.#balance; }
  get history() { return [...this.#history]; }  // return a copy
}

const b = new Bank();
b.deposit(100);
b.deposit(50);
b.withdraw(30);
console.log("balance:", b.balance);
console.log("history:", b.history);

// Outside access is a SyntaxError
try {
  console.log(b.#balance);
} catch (e) {
  console.log(e.name + ": " + e.message);
}

8. Getters and Setters

They look like normal properties from the outside but run code when read or written.

Code: 
class Temperature {
  #celsius = 0;

  // Getter: computed property
  get fahrenheit() {
    return this.#celsius * 9 / 5 + 32;
  }

  // Setter: validation + conversion
  set fahrenheit(f) {
    if (f < -459.67) throw new Error("below absolute zero");
    this.#celsius = (f - 32) * 5 / 9;
  }

  get celsius() { return this.#celsius; }
  set celsius(c) {
    if (c < -273.15) throw new Error("below absolute zero");
    this.#celsius = c;
  }
}

const t = new Temperature();
t.celsius = 25;
console.log(t.celsius, "°C =", t.fahrenheit.toFixed(1), "°F");

t.fahrenheit = 212;
console.log(t.celsius.toFixed(1), "°C =", t.fahrenheit, "°F");

try {
  t.celsius = -500;
} catch (e) {
  console.log(e.message);
}

9. Real World — Small Model Example

A CartItem class with a subclass DiscountedItem and a static factory.

Code: 
class CartItem {
  static TAX = 0.1;

  constructor(name, price, qty = 1) {
    this.name = name;
    this.price = price;
    this.qty = qty;
  }

  subtotal() {
    return this.price * this.qty;
  }

  total() {
    return Math.round(this.subtotal() * (1 + CartItem.TAX) * 100) / 100;
  }

  static fromJSON(json) {
    const d = typeof json === "string" ? JSON.parse(json) : json;
    return new CartItem(d.name, d.price, d.qty);
  }
}

class DiscountedItem extends CartItem {
  constructor(name, price, qty, percent) {
    super(name, price, qty);
    this.percent = percent;
  }
  subtotal() {
    // apply discount first
    const base = super.subtotal();
    return Math.round(base * (1 - this.percent / 100) * 100) / 100;
  }
}

const a = CartItem.fromJSON('{"name":"Pen","price":20,"qty":3}');
const b = new DiscountedItem("Laptop Bag", 1000, 1, 15);

console.log(a.name, "→ total", a.total());
console.log(b.name, "→ subtotal", b.subtotal(), "total", b.total());

// Static constants are shared by parent AND child
console.log(b instanceof CartItem);
console.log(b instanceof DiscountedItem);

10. Common Mistakes (and the errors they cause)

Code: 
class Animal {
  constructor(name) { this.name = name; }
  speak() { return "sound"; }
}

// ---- 1. Forgetting new ----
try {
  Animal("Rex");
} catch (e) {
  console.log("no new →", e.name);
}

// ---- 2. this before super in subclass ----
class Dog extends Animal {
  constructor(name) {
    // this.x = 1;      // ReferenceError
    super(name);
    this.x = 1;
  }
}

// ---- 3. Detached method ----
const d = new Dog("Rex");
const speak = d.speak;
try {
  speak();
} catch (e) {
  console.log("detached →", e.name);
}

// ---- 4. arrow as method ----
class Counter {
  count = 0;
  bad  = () => this.count;   // this = instance via class field
  good() { return this.count; }
}
const c = new Counter();
console.log(c.bad());
console.log(c.good());

// ---- 5. calling class static on an instance ----
class Math2 {
  static add(a, b) { return a + b; }
}
console.log(Math2.add(2, 3));
// new Math2().add(2, 3);   // TypeError

11. Quick Reference

Concept Syntax Note
Declare a class class Name { } Not hoisted — declare before use
Create an object new Name(args) Runs the constructor
Constructor constructor(a, b) { } Runs once on new
Instance field count = 0; Set on every new object
Static field static TAX = 0.1; Shared by all instances
Instance method total() { } this = instance
Static method static create() { } Name.method(), no instance
Getter get value() { } Reads like a property
Setter set value(v) { } Validates input
Private field #count = 0; Truly private (SyntaxError from outside)
Private method #log() { } Truly private
Inheritance class Child extends Parent { } Prototype chain setup
Parent constructor super(args) Must run before this
Parent method super.method() Extend instead of replace
Parent static super.staticMethod() Static chain
Type check obj instanceof Parent Walks the whole chain
Arrow as method method = () => this.x Uses class field binding
Attach method to callback obj.method.bind(obj) Or arrow wrapper
Class as a function typeof MyClass Returns "function"
Static on instance obj.staticMethod() Undefined — statics are class-only
