Core JavaScript Logic (State & Computation): 
// Calculator State Variables
let currentOperand = "";
let previousOperand = "";
let operation = undefined;

// 1. Append Digit or Decimal
function appendNumber(number) {
  if (number === "." && currentOperand.includes(".")) return;
  currentOperand = currentOperand.toString() + number.toString();
  updateDisplay();
}

// 2. Select Arithmetic Operation
function chooseOperation(op) {
  if (currentOperand === "") return;
  if (previousOperand !== "") compute();
  
  operation = op;
  previousOperand = currentOperand;
  currentOperand = "";
  updateDisplay();
}

// 3. Perform Arithmetic Computation
function compute() {
  let computation;
  const prev = parseFloat(previousOperand);
  const current = parseFloat(currentOperand);

  if (isNaN(prev) || isNaN(current)) return;

  switch (operation) {
    case "+": computation = prev + current; break;
    case "-": computation = prev - current; break;
    case "*": computation = prev * current; break;
    case "/": 
      computation = current === 0 ? "Error" : prev / current; 
      break;
    default: return;
  }

  currentOperand = computation;
  operation = undefined;
  previousOperand = "";
  updateDisplay();
}

2. Quick Reference

State / Action JavaScript Pattern Purpose
State Variable let currentOperand = "" Stores active number string displayed on screen
Append Digit currentOperand += num Concatenates digits as string before parsing
Prevent Extra Decimal if (num === "." && str.includes(".")) Prevents invalid inputs like "5.2.1"
BackSpace / Delete str.slice(0, -1) Removes the last character from input string
Convert to Number parseFloat(str) Converts input string to number before computation
Safe Computation switch (op) { case "+": ... } Computes math safely without using insecure eval()
Divide by Zero Guard current === 0 ? "Error" : prev / current Handles division by zero edge case cleanly
